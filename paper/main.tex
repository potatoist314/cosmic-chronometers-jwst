\section{Introduction}
\begin{itemize}
\item Chronometer: $H(z) = -\dfrac{1}{1+z}\dfrac{\mathrm{d}z}{\mathrm{d}t}$, from differential ages of passive galaxies \citep[eq.~2]{JimenezLoeb2002}.
\item Measurements: \citet{Simon2005}, \citet{Stern2010}, \citet{Moresco2012}, \citet{Moresco2015}, \citet{Moresco2016}. Review: \citet{Moresco2022}. Systematics: \citet{Moresco2018}, \citet{Moresco2020}.
\item Hubble tension:
  \begin{itemize}
  \item Planck 2018, base $\Lambda$CDM: $H_0 = 67.4\pm0.5$ \citep{Planck2020}.
  \item SH0ES Cepheid--SN baseline: $H_0 = 73.04\pm1.04$; ``5-sigma difference'' from Planck+$\Lambda$CDM \citep{Riess2022}.
  \end{itemize}
\item This work: Ceridwen \citep{Stoffers2026} fits of LEGA-C DR3 \citep{vanderWel2021} spectra and COSMOS photometry; mass-weighted SFH ages; $H(\zeff)$ at $\zeff\approx0.77$.
\end{itemize}

\section{Model and fitting}
\label{sec:model}
\src{wiki/notes/model.md}
\begin{itemize}
\item Code: Ceridwen \citep{Stoffers2026}. Sampler: nested slice sampling \citep{Yallup2026}, 500 live points, 65 inner steps.
\item SSPs: \afe-variable aMIST v2.5 isochrones, C3K v2.3 high-resolution library ($R\approx6000$), \citet{Kroupa2001} IMF; $\alpha$-MC grid \citep{Park2025}, 13 [Fe/H] $\times$ 5 \afe{} $\times$ 107 ages. Model detail only.
\item Priors: $\afe\sim U(-0.2,0.6)$; $[\mathrm{Fe/H}]\sim U(-2.5,0.5)$.
\item C/N responses: alf \citep{ConroyvanDokkum2012,Conroy2018}; $[\mathrm{C/Fe}]\sim U(-0.3,0.5)$, $[\mathrm{N/Fe}]\sim U(-0.3,1.0)$; clamped to age 1--13\,Gyr and [Fe/H] $-1.5$ to 0.
\item SFH: 14 continuity bins \citep{Leja2019}: 0--30\,Myr, 30--100\,Myr, log-spaced to $0.85\,t_\mathrm{univ}$, last bin to $t_\mathrm{univ}$. Prior: Student-$t$(0, 0.3\,dex, $\nu=2$) on 14 log SFR ratios.
  \todo{upstream Ceridwen 2.0 per-bin SFH (13 bin ratios, \citealt[eq.~4]{Leja2019}) merge is running; refit may change every number below}
\item \sig: $\mathcal{N}$(catalogue value, error), clipped at $\pm3$ errors.
\item Dust: \citet{Noll2009} law; birth-cloud dust on; $\tau\sim U(0,1)$.
\item Calibration: Chebyshev polynomial, order 10, marginalised; $f_\mathrm{calib}$ log-uniform 1--10\,per\,cent.
\item Fit window: rest 4074.5--4666.1\,\AA, the same for all galaxies. \src{results/dr3-more-galaxies-2026-11-16/REPORT.md}
\item Age: $t_\mathrm{MW} = \sum_i M_i\,\bar t_i / \sum_i M_i$, with $M_i$ the mass formed in bin $i$ and $\bar t_i$ its midpoint.
\item Estimators, least squares over galaxies:
  \begin{itemize}
  \item plain line: $t_\mathrm{MW} = a + b\,z$;
  \item \sig{} term: $t_\mathrm{MW} = a + b\,z + c\log_{10}\sig$.
  \end{itemize}
\item $H(\zeff) = -1/[(1+\zeff)\,b]$. $\zeff$: redshift where the reference $\mathrm{d}t/\mathrm{d}z$ equals the least-squares slope of reference ages at the sample redshifts (\texttt{effective\_redshift}, \src{src/chronometer.py}). 194: $\zeff = 0.775$; 333: 0.773. Range 0.774--0.775 for $H_0$ 60--75, $\Omega_\mathrm{m}$ 0.25--0.40, against median $z=0.728$. \src{wiki/research/questions/q-afe-age-chronometer.md}
\item Errors: galaxy bootstrap, 10000 draws, seed 20261110. \src{results/sigma-split-2026-11-16/sigma_split.py}
\end{itemize}
